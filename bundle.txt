package com.keystudio.kb

import android.Manifest
import android.content.ClipboardManager
import android.content.Context
import android.content.Intent
import android.content.pm.PackageManager
import android.inputmethodservice.InputMethodService

import android.speech.RecognizerIntent
import android.speech.SpeechRecognizer
import android.view.KeyEvent
import android.view.View
import android.view.inputmethod.EditorInfo

class KeyService : InputMethodService(), KbListener {
    private var view: KeyboardView? = null
    private var rec: SpeechRecognizer? = null

    override fun onCreateInputView(): View {
        val v = KeyboardView(this)
        v.listener = this
        v.reload()
        view = v
        return v
    }

    override fun onEvaluateFullscreenMode(): Boolean = false

    override fun onStartInputView(info: EditorInfo?, restarting: Boolean) {
        super.onStartInputView(info, restarting)
        view?.reload()
    }

    override fun onFinishInputView(finishingInput: Boolean) {
        super.onFinishInputView(finishingInput)
        stopVoice()
    }

    override fun onText(s: String) {
        currentInputConnection?.commitText(s, 1)
    }

    override fun onDelete() {
        sendDownUpKeyEvents(KeyEvent.KEYCODE_DEL)
    }

    override fun onEnter() {
        val opts = currentInputEditorInfo?.imeOptions ?: 0
        val a = opts and EditorInfo.IME_MASK_ACTION
        val noEnter = (opts and EditorInfo.IME_FLAG_NO_ENTER_ACTION) != 0
        if (!noEnter && a != EditorInfo.IME_ACTION_NONE && a != EditorInfo.IME_ACTION_UNSPECIFIED) {
            currentInputConnection?.performEditorAction(a)
        } else {
            sendDownUpKeyEvents(KeyEvent.KEYCODE_ENTER)
        }
    }

    override fun onSettings() {
        startActivity(Intent(this, SettingsActivity::class.java).addFlags(Intent.FLAG_ACTIVITY_NEW_TASK))
    }

    override fun getClips(): List<String> {
        val p = getSharedPreferences("ks", 0)
        val list = ArrayList<String>()
        val saved = p.getString("clips", "") ?: ""
        if (saved.isNotEmpty()) list.addAll(saved.split("\u0001"))
        try {
            val cm = getSystemService(Context.CLIPBOARD_SERVICE) as ClipboardManager
            val t = cm.primaryClip?.getItemAt(0)?.text?.toString()
            if (!t.isNullOrBlank()) {
                list.remove(t)
                list.add(0, t)
            }
        } catch (e: Exception) {
        }
        val out = list.take(20)
        p.edit().putString("clips", out.joinToString("\u0001")).apply()
        return out
    }

    private fun stopVoice() {
        rec?.destroy()
        rec = null
        view?.listening = false
    }

    override fun onVoice() {
        if (rec != null) {
            stopVoice()
            return
        }
        if (checkSelfPermission(Manifest.permission.RECORD_AUDIO) != PackageManager.PERMISSION_GRANTED) {
            startActivity(
                Intent(this, SettingsActivity::class.java)
                    .addFlags(Intent.FLAG_ACTIVITY_NEW_TASK).putExtra("perm", true)
            )
            return
        }
        if (!SpeechRecognizer.isRecognitionAvailable(this)) return
        val r = SpeechRecognizer.createSpeechRecognizer(this)
        r.setRecognitionListener(object : RecognitionListener {
            override fun onReadyForSpeech(params: Bundle?) {}
            override fun onBeginningOfSpeech() {}
            override fun onRmsChanged(rmsdB: Float) {}
            override fun onBufferReceived(buffer: ByteArray?) {}
            override fun onEndOfSpeech() {}
            override fun onError(error: Int) { stopVoice() }
            override fun onPartialResults(partialResults: Bundle?) {}
            override fun onEvent(eventType: Int, params: Bundle?) {}
            override fun onResults(results: Bundle?) {
                val t = results?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)?.firstOrNull()
                if (!t.isNullOrBlank()) onText("$t ")
                stopVoice()
            }
        })
        val tag = mapOf("ru" to "ru-RU", "en" to "en-US", "uk" to "uk-UA", "de" to "de-DE")[view?.lang ?: "ru"] ?: "ru-RU"
        val i = Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH)
        i.putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM)
        i.putExtra(RecognizerIntent.EXTRA_LANGUAGE, tag)
        rec = r
        view?.listening = true
        r.startListening(i)
    }

    override fun onDestroy() {
        stopVoice()
        super.onDestroy()
    }
}

        fun place(list: ArrayList<Key>, y: Float, specs: List<Spec>) {

                if (s.code.isNotEmpty()) {

                    k.x = pad + col * (colW + gap)
                    k.y = y
                    k.w = s.span * colW + (s.span - 1) * gap

                    val pd = (24 - 2 * n - 2 * side) / 2

                    if (last) specs.add(Spec("⌫", "bs", side, false, 1))
                    if (pd > 0) specs.add(Spec("", "", pd))
                    place(keys, panelTop + i * (kh + gap), specs)
